cmake_minimum_required(VERSION 3.16)
project(prime_native_recorder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_JIT OFF CACHE BOOL "" FORCE)
set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
if(NOT MELON_SOURCE)
  message(FATAL_ERROR "Set MELON_SOURCE to the pinned melonPrimeDS checkout")
endif()
execute_process(COMMAND git -C "${MELON_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE MELON_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT MELON_COMMIT STREQUAL "41b087835978cb99e0d1b862411b702c5d1648c8")
  message(FATAL_ERROR "The recorder requires the documented melonPrimeDS commit")
endif()
execute_process(COMMAND git -C "${MELON_SOURCE}" diff --quiet HEAD RESULT_VARIABLE SOURCE_DIRTY)
if(NOT SOURCE_DIRTY EQUAL 0)
  message(FATAL_ERROR "The reference emulator checkout must have no tracked modifications")
endif()
list(APPEND CMAKE_MODULE_PATH "${MELON_SOURCE}/cmake")
add_subdirectory(${MELON_SOURCE} melon)
# Instrument an out-of-tree copy of the pinned interpreter. The emulator
# checkout remains pristine; JIT is disabled so every observation is reached.
file(READ "${MELON_SOURCE}/src/ARM.cpp" ARM_SOURCE)
set(PROBE_NEEDLE "            // actually execute\n            if (CheckCondition(CurInstr >> 28))")
string(FIND "${ARM_SOURCE}" "${PROBE_NEEDLE}" PROBE_POSITION)
if(PROBE_POSITION EQUAL -1)
  message(FATAL_ERROR "Pinned ARM interpreter observation point not found")
endif()
string(REPLACE "${PROBE_NEEDLE}" "            if (Num == 0 && R[15] >= 0x0201f000 && R[15] <= 0x02021c08) PrimeNativeProbe(R[15] - 8, R);\n${PROBE_NEEDLE}" ARM_SOURCE "${ARM_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ARM-probed.cpp" "#include \"probe.h\"\n${ARM_SOURCE}")
get_target_property(CORE_SOURCES core SOURCES)
list(FILTER CORE_SOURCES EXCLUDE REGEX "(^|/)ARM[.]cpp$")
set_property(TARGET core PROPERTY SOURCES "${CORE_SOURCES}")
target_sources(core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ARM-probed.cpp")
target_include_directories(core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${MELON_SOURCE}/src")
add_executable(prime-native bridge.cpp platform.cpp probe.cpp)
target_include_directories(prime-native PRIVATE ${MELON_SOURCE}/src)
target_link_libraries(prime-native PRIVATE core ${CMAKE_DL_LIBS})
