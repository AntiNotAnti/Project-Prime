cmake_minimum_required(VERSION 3.20)
project(PrimeRmlUiTextEditCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PRIME_RMLUI_NATIVE_LIBRARY}" OR NOT EXISTS "${RMLUI_SOURCE_DIR}/Include/RmlUi/Core.h")
    message(FATAL_ERROR "Pass PRIME_RMLUI_NATIVE_LIBRARY and the pinned RMLUI_SOURCE_DIR; this tool does not build/download either.")
endif()
get_filename_component(PRIME_RMLUI_NATIVE_DIRECTORY "${PRIME_RMLUI_NATIVE_LIBRARY}" DIRECTORY)
add_executable(PrimeTextEditCpuCheck main.cpp)
target_compile_definitions(PrimeTextEditCpuCheck PRIVATE RMLUI_STATIC_LIB)
target_include_directories(PrimeTextEditCpuCheck PRIVATE "../../../native/rmlui-poc" "${RMLUI_SOURCE_DIR}/Include" "${RMLUI_SOURCE_DIR}/Source/Core")
target_link_libraries(PrimeTextEditCpuCheck PRIVATE "${PRIME_RMLUI_NATIVE_LIBRARY}")
set_target_properties(PrimeTextEditCpuCheck PROPERTIES BUILD_RPATH "${PRIME_RMLUI_NATIVE_DIRECTORY}")
