using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Collections.Generic;
using System.Text.Json;
using MphRead.Mods.Network;
using OpenTK.Mathematics;
namespace MphRead.Mods.Replay;
internal static class ReplayV134FixtureExport
{
    internal static int Run(string output)
    {
        Directory.CreateDirectory(output);
        const BindingFlags flags = BindingFlags.Static | BindingFlags.NonPublic;
        var codec = typeof(ReplayWorldCheckpoint);
        var types = (Dictionary<string, Type>)codec.GetField("Types", flags)!.GetValue(null)!;
        var objects = (Type[])codec.GetField("ObjectTypes", flags)!.GetValue(null)!;
        var valueFields = codec.GetMethod("ValueFields", flags)!;
        var manifest = new {
            commit = "f0e01e09e08467974b8a5ea359d690ebe0a7e3a4",
            contract = (string)codec.GetField("Contract", flags)!.GetValue(null)!,
            objectTypes = objects.Select(t => t.ToString()).ToArray(),
            valueFields = types.Values.Where(t => t.IsValueType && !t.IsPrimitive && !t.IsEnum)
                .ToDictionary(t => t.ToString(), t => ((FieldInfo[])valueFields.Invoke(null, new object[] { t })!).Select(f => f.Name).ToArray())
        };
        File.WriteAllText(Path.Combine(output, "schema-v134.json"), JsonSerializer.Serialize(manifest, new JsonSerializerOptions { WriteIndented = true }));
        Console.WriteLine("Historical contract " + manifest.contract);
        Paths.UpdatePaths(); Paths.ChooseMphPath(); Headless.Enter();
        string recording = Path.Combine(output, "v134-source.ppdemo");
        ReplayWorldCoverageCheck.Write(recording, "MP1 SANCTORUS", GameMode.Battle, new Vector3(0, 2, 0));
        using var replay = new PassiveReplayScene(recording, new(256, 192));
        for (int i = 0; i < 120; i++) if (!replay.Step()) throw new Exception("Fixture ended early.");
        using var checkpoint = ReplayWorldCheckpoint.Capture(replay);
        File.WriteAllBytes(Path.Combine(output, "v134-world.ppwc"), checkpoint.Bytes.ToArray());
        var expected = new { frame = checkpoint.Frame, points = replay.Scene.GameState.Points,
            players = replay.Scene.Players.Items.Select(p => new { slot = p.SlotIndex,
                position = new[] { p.Position.X, p.Position.Y, p.Position.Z },
                speed = new[] { p.Speed.X, p.Speed.Y, p.Speed.Z },
                facing = new[] { p.FacingVector.X, p.FacingVector.Y, p.FacingVector.Z },
                health = p.Health, flags1 = (uint)p.Flags1, flags2 = (uint)p.Flags2 }).ToArray() };
        File.WriteAllText(Path.Combine(output, "v134-expected.json"), JsonSerializer.Serialize(expected, new JsonSerializerOptions { WriteIndented = true }));
        Console.WriteLine($"Historical checkpoint frame={checkpoint.Frame} bytes={checkpoint.Bytes.Length}");
        return 0;
    }
}
